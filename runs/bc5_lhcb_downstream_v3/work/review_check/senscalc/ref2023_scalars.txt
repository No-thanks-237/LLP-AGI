----- cell 6  label=Scalar phenomenology -----
{MP=0.938;
(*Scalar-nucleon low-energy coupling*), 
, GSNN=1.2*10^-3;
(*Fit for the pp total cross-section in mb in terms of invariant mass s*), 
, Zfit=35.45;, 
, Bfit=0.308;, 
, Y1fit=42.53;, 
, Y2fit=33.34;, 
, s0fit=5.38^2;, 
, s1fit=1;, 
, \[Eta]1fit=0.458;, 
, \[Eta]2fit=0.545;, 
, S[Pp_,mp_]=2*Pp*mp;, 
, Sprime[Pp_,mp_,z_]=2*mp*Pp(1-z)+2*mp^2;, 
, sigmapp[s_]=Zfit+Bfit*Log[s/s0fit]^2+Y1fit*(s1fit/s)^\[Eta]1fit-Y2fit(s1fit/s)^\[Eta]2fit;
(*Kinematics of the 1 -> 2 process p -> p + S within the WW approximation*), 
, EPbrem=Pp+(mp^2)/(2*Pp);, 
, EpprimeBrem=Pp*(1-z)+((mp^2+pT^2))/(2*Pp(1-z));, 
, ppprimeBrem=(1-z)*Pp;, 
, ESbrem=Pp*z+((pT^2+mS^2))/(2*Pp*z);, 
, pSbrem=pp-ppprimeBrem;, 
, Print["Applicability parameter (must be << 1):"], 
, ApplicabilityParameter[Pp_,z_,pT_,mX_,mp_]=Simplify[((EpprimeBrem+ESbrem-EPbrem))/(((EpprimeBrem-ESbrem+EPbrem)/.{pT->0,mp->0,mX->0}))]/.{mS->mX}/.{mX^2+4 Pp^2 (z-1) z->4Pp^2z(z-1)}
(*Momenta of the protons at various facilities*), 
, {Ppfacility["SPS"],Ppfacility["LHC"],Ppfacility["FCC-hh"],Ppfacility["FermilabBD"]}={400.,7000.,50000.,120.};
(*The minimal allowed fraction of energy E_S/E_p*), 
, {zMinBrem["FermilabBD"],zMinBrem["SPS"],zMinBrem["LHC"],zMinBrem["FCC-hh"]}={0.1,0.1,0.005,0.0001};
(*The maximal allowed fraction of energy*), 
, zMaxBrem=0.9;
(*Elastic SNN form-factor*), 
, BremFormFactorSquared[mS_]=Interpolation[Import[FileNameJoin[{NotebookDirectory[],"phenomenology/Scalar/branching ratios/proton-timelike-form-factor-scalar.txt"}],"Table"],InterpolationOrder->1][mS];}

{<<FeynCalc`
(*Matrix element corresponding to the sub-process p -> p + S*), 
, MscalarBrem=gSNN SpinorUBar[p1,mp].SpinorU[k1,mp];, 
, MscalarBremStar=ComplexConjugate[MscalarBrem];
(*Squared matrix element*), 
, MScalarBremSquared=((1)/(2)FermionSpinSum[MscalarBrem MscalarBremStar]/.DiracTrace->TR//Contract//Simplify)/.{ScalarProduct[k1,p1]-> Pp*EpprimeBrem-Pp*ppprimeBrem};
(*Differential splitting probability*), 
, DifferentialSplittingProbability[mp_,mS_,pT_,z_,gSNN_,Pp_]=Simplify[((1)/(16*Pi^2*z)(1)/(4*Pp*EpprimeBrem)*(MScalarBremSquared)/((EpprimeBrem+ESbrem-Pp)^2))]/.{ (mp^2+2 Pp^2) ->2*Pp^2}/.{mp^2+2 Pp^2 (z-1)^2+pT^2->2*Pp^2(z-1)^2};, 
, pTmaxBrem=DialogInput[{br=""},Column[{"Enter the maximal \!\(\*SubscriptBox[\(p\), \(T\)]\) (in GeV) for the bremsstrahlung channel (typical value is 1 GeV):",InputField[Dynamic[br],Number],Button["Proceed",DialogReturn[br],ImageSize->Automatic]}]]//N;
(*Double distribution in theta and E*), 
, DistrBremInpTz[mS_,pT_,z_,Pp_,zmin_]=DifferentialSplittingProbability[MP,mS,pT,z,GSNN,Pp]*2*pT*(sigmapp[Sprime[Pp,MP,z]])/(sigmapp[S[Pp,MP]])UnitStep[pTmaxBrem-pT]*UnitStep[z-zmin]UnitStep[zMaxBrem-z];, 
, pTinthetaE[thetaS_,ES_,mS_]=Sqrt[ES^2-mS^2]Sin[thetaS];, 
, zInthetaE[ES_,Pp_]=(ES)/(Pp);, 
, JacobianpTzToESthetaS[thetaS_,ES_]=D[pTinthetaE[thetaS,ES,mS],thetaS]*D[zInthetaE[ES,Pp],ES];, 
, DistrBremthetaEscalarNoFormFactorTemp[mS_,thetaS_,ES_,Pp_,zmin_]=DistrBremInpTz[mS,pTinthetaE[thetaS,ES,mS],zInthetaE[ES,Pp],Pp,zmin]*JacobianpTzToESthetaS[thetaS,ES];, 
, {mSminFromBrem,mSmaxFromBrem}={0.02,2.5};, 
, ESmaxBremBeam[Facility_,thetaS_]:=Min[zMaxBrem*Ppfacility[Facility],(pTmaxBrem)/(Sin[thetaS]+10^-7)];, 
, thetamaxBremTemp[Facility_]:=(thetaS/.Solve[zMinBrem[Facility]*Ppfacility[Facility]==(pTmaxBrem)/(Sin[thetaS]),thetaS][[1]]/.{TemplateBox[{1}, C]->0});, 
, {thetamaxBremFacility["FermilabBD"],thetamaxBremFacility["SPS"],thetamaxBremFacility["LHC"],thetamaxBremFacility["FCC-hh"]}=thetamaxBremTemp[#]&/@{"FermilabBD","SPS","LHC","FCC-hh"};, 
, NormalizationBremNoFormFactorTemp[Facility_]:=Module[{zmin,ESmaxfrombrem,pbeam,thetamaxbrem},
pbeam=Ppfacility[Facility];
zmin=zMinBrem[Facility];
ESmaxfrombrem[thetaS_]=ESmaxBremBeam[Facility,thetaS];
thetamaxbrem=thetamaxBremFacility[Facility];
Interpolation[Table[{mS,NIntegrate[DistrBremthetaEscalarNoFormFactorTemp[mS,thetaS,ES,pbeam,zmin],{thetaS,0,thetamaxbrem},{ES,zmin*pbeam,Max[zmin*pbeam,ESmaxfrombrem[thetaS]]},Method->"AdaptiveMonteCarlo"]},{mS,mSminFromBrem,mSmaxFromBrem,((mSmaxFromBrem-mSminFromBrem))/(20)}],InterpolationOrder->1][mS]
], 
, Do[NormalizationBremNoFormFactor[mS_,Facility]=NormalizationBremNoFormFactorTemp[Facility];
DoubleDistrScalarfromBremFacility[mS_,thetaS_,ES_,Facility]=If[mSminFromBrem<=mS<=mSmaxFromBrem,Evaluate[(1)/(NormalizationBremNoFormFactor[mS,Facility])*DistrBremthetaEscalarNoFormFactorTemp[mS,thetaS,ES,Ppfacility[Facility],zMinBrem[Facility]]],0];
PbremFacility[mS_,theta2_,Facility]=theta2*BremFormFactorSquared[mS]*NormalizationBremNoFormFactor[mS,Facility];
ESmaxFromBrem[mS_,thetaS_,Facility]=ESmaxBremBeam[Facility,thetaS],{Facility,{"FermilabBD","SPS","LHC","FCC-hh"}}];
(*Tabulated distribution to be used in next sections*), 
, TableBremDistr[thetaminExp_,thetamaxExp_,Facility_]:=Module[{eminbrem,thetaSlistbrem,EslistBremTemp,EslistBrem},
eminbrem=Ppfacility[Facility]*zMinBrem[Facility];
thetaSlistbrem=Table[thetaV,{thetaV,0.99thetaminExp,1.01thetamaxExp,(1.01thetamaxExp-0.99thetaminExp)/50}];
EslistBremTemp=Join[Table[x,{x,0.02,eminbrem,1}],Table[10^x,{x,Log10[1.02*eminbrem],Log10[Ppfacility[Facility]],((Log10[Ppfacility[Facility]]-Log10[1.02*eminbrem]))/(50)}]];
EslistBrem=Join[EslistBremTemp,ESmaxBremBeam[Facility,#]&/@thetaSlistbrem]//Sort//DeleteDuplicates;
Flatten[Table[{mS,thetaS,ES,DoubleDistrScalarfromBremFacility[mS,thetaS,ES,Facility]+10^-90},{mS,mSminFromBrem,mSmaxFromBrem//N,((mSmaxFromBrem//N)-mSminFromBrem)/10},{thetaS,thetaSlistbrem},{ES,EslistBrem}],{1,2,3}]
]}
