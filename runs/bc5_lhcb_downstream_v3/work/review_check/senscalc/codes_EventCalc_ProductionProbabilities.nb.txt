MotherFraction["h"]=PppTohExperiment[GivenExperiment];

MotherFraction["Upsilon"]=PppToUpsilonFacility[Facility];
MotherFraction["Fragmentation"]=MotherFraction["Old-Mixing"]=1.;
,{Facility,{FacilitySelected}}]

Br[Eta]prToV[mLLP_]=If[mLLP<mSM["EtaPr"],Evaluate[2*(1-mLLP^2/mSM["EtaPr"]^2)^3*Br[Eta]prTogammagamma],0];
{BrDecay[mLLP_,epsilon2_,"Pi0"],BrDecay[mLLP_,epsilon2_,"Eta"],BrDecay[mLLP_,epsilon2_,"EtaPr"]}=epsilon2*{BrPi0ToV[mLLP],Br[Eta]ToV[mLLP],Br[Eta]prToV[mLLP]};
Do[
PmotherToLLP[mLLP_,epsilon2_,coupling2_,part]=BrDecay[mLLP,epsilon2,part];
,{part,{"Pi0","Eta","EtaPr"}}]
]

If[LLPdirName=="DP",
(*_____________________________*)
(*Production from Drell-Yan process*)
(*_____________________________*)
(*Average enhancement LO vs LO+NLO obtained for SPS energies in 2011.05115*)
kfactorNLO=1.7;
[Sigma]DrellYanData[Facility_]:=Block[{},
dat=Import[FileNameJoin[{directory["Production weights"],"sigmaDrellYan_DP_"<>Facility<>".txt"}],"Table"];
MinMaxMassesDrellYan[Facility]={Min[dat[[All,1]]],Max[dat[[All,1]]]};
{int[ma_,"Central"],int[ma_,"Upper"],int[ma_,"Lower"]}=Table[If[MinMaxMassesDrellYan[Facility][[1]]<= ma<=MinMaxMassesDrellYan[Facility][[2]],Evaluate[10^(Interpolation[{Log10[#[[1]]],Log10[#[[i]]+10^-90]}&/@dat,InterpolationOrder->1][Log10[ma]])],0.],{i,2,4}];
(*k factor from *)
{PmotherToLLPTemp[mLLP_,epsilon2_,"DrellYan-Central",Facility],PmotherToLLPTemp[mLLP_,epsilon2_,"DrellYan-Upper",Facility],PmotherToLLPTemp[mLLP_,epsilon2_,"DrellYan-Lower",Facility]}=If[Facility=="SPS",kfactorNLO,1]*epsilon2*(int[mLLP,#]&/@{"Central","Upper","Lower"})/([Sigma]pNucleonInpb[AtargetSelected]);
];
If[MemberQ[{"SPS","FermilabBD","LHC"},FacilitySelected],
[Sigma]DrellYanData[FacilitySelected];
drellyanunc=dropdownDialog[{"Lower","Central","Upper"},"Select the variation of Drell-Yan flux within uncertainty band:"];
PmotherToLLP[mLLP_,epsilon2_,coupling2_,"DrellYan"]=PmotherToLLPTemp[mLLP,epsilon2,"DrellYan-"<>drellyanunc,FacilitySelected];
];
]

(*________________*)
(*Bremsstrahlung*)
(*________________*)
BremProd[Facility_]:=Module[{dbr,bremprobfiles,bremprobfilenames},
bremprobfiles=FileNames["ProdProb*_"<>Facility<>".m",directory["Production weights"]];
If[bremprobfiles!={},
(*Scanning over all available files with the bremsstrahlung probability. They describe different descriptions of the bremsstrahlung*)
(*Extract the bremsstrahlung description from the filename*)
bremprobfilenames=Last@FileNameSplit[#]&/@bremprobfiles;
bremdescriptions=(StringSplit[#,{"ProdProb_"<>LLPdirName<>"_","_"}][[1]])&/@bremprobfilenames;
bremdescription=dropdownDialog[bremdescriptions,"Select the description of the DP production via proton bremsstrahlung:"];
bremprobfiles=Select[bremprobfiles,StringContainsQ[#,bremdescription]&][[1]];
bR="Bremsstrahlung";
datbrem[mLLP_,thetaLLP_]=Import[bremprobfiles,"MX"];
If[StringContainsQ[bremdescription,"Baseline"],
PmotherToLLP[mLLP_,epsilon2_,coupling2_,bR]=datbrem[mLLP,thetaLLP][[1]]*epsilon2;
ELLPmin[bR]=Evaluate[datbrem[mLLP,thetaLLP][[2]]];
thetamaxbrem[bR]=Evaluate[datbrem[mLLP,thetaLLP][[3]]];
ELLPmax[mLLP_,thetaLLP_,bR]=Evaluate[datbrem[mLLP,thetaLLP][[4]]];
,
bremUncertainties=datbrem[mLLP,thetaLLP][[All,1]];
bremunc=dropdownDialog[bremUncertainties,"Select the variation of bremsstrahlung within uncertainty band:"];
dbr[mLLP_,thetaLLP_]=Select[datbrem[mLLP,thetaLLP],#[[1]]==bremunc&][[1]];
PmotherToLLP[mLLP_,epsilon2_,coupling2_,bR]=dbr[mLLP,thetaLLP][[2]]*epsilon2;
ELLPmin[bR]=Evaluate[dbr[mLLP,thetaLLP][[3]]];
thetamaxbrem[bR]=Evaluate[dbr[mLLP,thetaLLP][[4]]];
ELLPmax[mLLP_,thetaLLP_,bR]=Evaluate[dbr[mLLP,thetaLLP][[5]]];
];
];
];
BremProd[FacilitySelected]
]

Print["Description of the production via mixing/fragmentation"];
SelectedMixingChoice=selectionDialogExplanationOneOption[{"Revised (2409.11096)","Old (1810.01879)"},{"Production incorporated directly in the fragmentation chain in PYTHIA8 (2409.11096)",LLPdirName<>" flux = meson flux * squared mixing angle (aka 1810.01879 and refs. therein)"},"Select the description of the "<>LLPdirName<>" production via fragmentation:",200];
condrev=SelectedMixingChoice=="Revised (2409.11096)";
If[condrev,
YieldsFragmentation[mLLP_]=Import[FileNameJoin[{directory["Production weights"],"Yield-Fragmentation_DP.mx"}],"MX"];
(*Old description a-la 2201.05170*)
Print[SelectedMixingChoice];
Do[
If[YieldsFragmentation[mLLP][[i]][[1]]==FacilitySelected,
{mmixingmin,mmixingmax}=YieldsFragmentation[mLLP][[1]][[2]][[1]][[3]][[0]][[1]][[1]];
PmotherToLLP[mLLP_,epsilon2_,coupling2_,"Fragmentation"]=epsilon2*If[mmixingmin<=mLLP<=mmixingmax,Evaluate[YieldsFragmentation[mLLP][[i]][[2]]],0.];
]
,{i,1,Length[YieldsFragmentation[mLLP]],1}];
,
(*Mixing angle [Rho]^0-DP. Following https://arxiv.org/pdf/1810.01879.pdf*)
g[Rho]0=5;
thetaDP[Rho]02[mLLP_,epsilon2_]=epsilon2*(4Pi*alphaEM)/(g[Rho]0^2)(mSM["Rho0"]^4)/((mLLP^2-mSM["Rho0"]^2)^2+[CapitalGamma]SM["[Rho]780"]^2*mSM["Rho0"]^2);
PmotherToLLP[mLLP_,epsilon2_,coupling2_,"Old-Mixing"]=Yield[Rho]0*thetaDP[Rho]02[mLLP,epsilon2]/.{Yield[Rho]0->MotherFraction["Rho0"]};
];
]
