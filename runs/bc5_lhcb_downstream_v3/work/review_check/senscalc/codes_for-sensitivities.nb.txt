(*Function to process the table*)
FindingMaxEnergy[tab_,order_]:=Module[{Egrouped,tot,thetasummed,accum,sel,emaxpos,emaxval},
(*Group by y values*)
Egrouped=GatherBy[tab,#[[2]]&];
(*Sum z values for each y and create new table*)
thetasummed=Table[{Egrouped[[i,1,2]],Total[Egrouped[[i,All,3]]]},{i,1,Length[Egrouped]}];
If[Max[thetasummed[[All,2]]]>10^-30.,
tot=thetasummed[[All,2]]//Total;
accum=(tot-Accumulate[thetasummed[[All,2]]])/(tot);
sel=Select[accum,#>=10^-order&];
If[Length[sel]!=0.,
emaxpos=Min[Position[accum,sel[[-1]]][[1]][[1]]+1,Length[accum]];
emaxval=thetasummed[[emaxpos]][[1]],
emaxval=thetasummed[[1]][[1]]];
,
emaxval=thetasummed[[1]][[1]];
];
emaxval
]
FindingEmaxtheta[mass_,data0_,Facility_,order_]:=Module[{thetaminmax,eminmax,GridIn,distrv,GridOut,xigDistr,distrvals,mappeddistr,thetaemax,emax,thetaranges,distrdatalog,ColliderOption,data},
data=Select[data0,#[[1]]==mass&&#[[3]]>mass&][[All,{2,3,4}]];
distrdatalog=DataLogarithmiedComp[data];
GridIn=(DeleteDuplicates[distrdatalog[[All,#]]]&/@{1,2})//N;
distrv=ArrayReshape[distrdatalog[[All,3]],{Length[GridIn[[1]]],Length[GridIn[[2]]]}];
thetaminmax=MinMax[data[[All,1]]];
eminmax=MinMax[data[[All,2]]];
ColliderOption=MemberQ[{"LHC","FCC-hh"},Facility];
thetaranges=If[ColliderOption,Select[{thetaminmax[[1]],10^-4.,10^-3.,5.*10^-3,10^-2.,3*10^-2.,6*10^-2.,10^-1.,2*10^-1.,5*10^-1,6*10^-1,1.,1.3,(Pi)/(2)},thetaminmax[[1]]<=#&],Select[Flatten[Join[{#,Pi-#}&/@{thetaminmax[[1]],10^-4.,10^-3.,5.*10^-3,10^-2.,3*10^-2.,6*10^-2.,10^-1.,2*10^-1.,5*10^-1,6*10^-1,1.,1.3,(Pi)/(2.),thetaminmax[[2]]}]]//N//Sort//DeleteDuplicates,thetaminmax[[1]]<=#<=thetaminmax[[2]]&]]//N;
thetaemax={{}};
Do[
GridOut={Table[th,{th,thetaranges[[i]],thetaranges[[i+1]],(thetaranges[[i+1]]-thetaranges[[i]])/(20)}],Join[Table[Exp[e],{e,Log[eminmax[[1]]],Log[eminmax[[2]]],Max[0.1,(Log[eminmax[[2]]]-Log[eminmax[[1]]])/(100)]}],{eminmax[[2]]}]//DeleteDuplicates}//Log;
xigDistr=MapThread[Interpolation[Transpose@{#,Range@Length@#},InterpolationOrder->1][#2]&,{GridIn,GridOut}];
distrvals=Flatten@cf2[Sequence@@xigDistr,distrv];
mappeddistr=Exp[Join[Tuples[GridOut],Partition[distrvals,1],2]];
emax=FindingMaxEnergy[mappeddistr,order];
thetaemax=Join[thetaemax,{{(thetaranges[[i]]+thetaranges[[i+1]])/(2),emax}}];
,{i,1,Length[thetaranges]-1,1}];
thetaemax=Drop[thetaemax,1];
thetaemax={mass,#[[1]],Max[mass,#[[2]]]}&/@(If[ColliderOption,Flatten[Join[{#,{Pi-#[[1]],#[[2]]}&/@#}&/@{Join[{{thetaminmax[[1]],thetaemax[[1]][[2]]}},thetaemax]}][[1]],1],Join[{{thetaminmax[[1]],thetaemax[[1]][[2]]}},thetaemax,{{thetaminmax[[2]],thetaemax[[-1]][[2]]}}]]//Sort//DeleteDuplicates//N)
]

]
